from flask import Flask, render_template, request, redirect, session from datetime import datetime, timedelta import os app = Flask(__name__) app.secret_key = os.environ.get("SECRET_KEY", "ganti-secret-key-ini") USERNAME = "admin" PASSWORD = "1234" # Untuk testing gunakan 60 detik. # Kalau mau 24 jam: 24 * 60 * 60 TIMEOUT_SECONDS = 60 last_checkin = None timeout_triggered = False def timeout_action(): """ Aksi yang dijalankan ketika timer habis. Untuk sekarang hanya mencetak pesan. """ print("\n⚠️ CHECK-IN TIMEOUT") print("User tidak melakukan check-in tepat waktu.\n") @app.route("/", methods=["GET", "POST"]) def index(): global last_checkin, timeout_triggered if request.method == "POST": username = request.form.get("username") password = request.form.get("password") if username == USERNAME and password == PASSWORD: session["logged_in"] = True # Reset timer last_checkin = datetime.now() timeout_triggered = False return redirect("/") return render_template( "index.html", error="Username atau password salah." ) # Belum login if not session.get("logged_in"): return render_template("index.html", logged_in=False) # Cek timer remaining = 0 if last_checkin: elapsed = (datetime.now() - last_checkin).total_seconds() remaining = max(0, TIMEOUT_SECONDS - int(elapsed)) if remaining == 0 and not timeout_triggered: timeout_triggered = True timeout_action() return render_template( "index.html", logged_in=True, remaining=remaining, timeout=timeout_triggered ) @app.route("/checkin", methods=["POST"]) def checkin(): global last_checkin, timeout_triggered if not session.get("logged_in"): return redirect("/") last_checkin = datetime.now() timeout_triggered = False return redirect("/") @app.route("/logout") def logout(): session.clear() return redirect("/") if __name__ == "__main__": app.run(host="127.0.0.1", port=8080, debug=False)
